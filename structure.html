Question 1: Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form ?

Solution
#include < stdio.h >
struct Employee {
    int empID;
    char name[50];
    char designation[50];
};

int main() {
    struct Employee emp[50];
    int i;

    printf("--- Enter Details for 50 Employees ---\n\n");

    for(i = 0; i < 50; i++) {
        printf("Entering details for Employee %d:\n", i + 1);
        
        printf("Enter Employee ID: ");
        scanf("%d", &emp[i].empID);
       
         printf("Enter Name: ");
        scanf("%49", emp[i].name); 
        
         printf("Enter Designation: ");
        scanf("%49", emp[i].designation);
        
        printf("-----------------------------------\n");
    }

    printf("\n======================================================\n");
    printf("%-10d | %-20s | %-20s\n", "Emp ID", "Name", "Designation");
    printf("======================================================\n");

    for(i = 0; i < 50; i++) {
        printf("%-10d | %-20s | %-20s\n", emp[i].empID, emp[i].name, emp[i].designation);
    }
    
    printf("======================================================\n");
}
